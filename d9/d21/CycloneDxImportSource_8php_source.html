38  public function parse()
39  {
40  $content = file_get_contents($this->filename);
41  if ($content === false) {
42  return false;
43  }
44 
45  $this->bomData = json_decode($content, true);
46  if (!is_array($this->bomData)) {
47  return false;
48  }
49 
50  if (!isset($this->bomData['bomFormat']) ||
51  strtolower($this->bomData['bomFormat']) !== 'cyclonedx') {
52  return false;
53  }
54 
55  if (isset($this->bomData['components']) && is_array($this->bomData['components'])) {
56  foreach ($this->bomData['components'] as $index => $component) {
57  if (!is_array($component)) {
58  continue;
59  }
60  $bomRef = $this->getComponentKey($component, $index);
61  $this->componentsByBomRef[$bomRef] = $component;
62  }
63  }
64 
65  return true;
66  }

136  public function getDataForFile($fileid)
137  {
138  if (!isset($this->componentsByBomRef[$fileid])) {
139  return new ReportImportData();
140  }
141 
142  $component = $this->componentsByBomRef[$fileid];
143 
144  $concludedLicenses = [];
145  $infoInFileLicenses = [];
146  $copyrightTexts = [];
147 
148  if (isset($component['licenses']) && is_array($component['licenses'])) {

150  $items = $this->parseLicenseEntry($licenseEntry);
151  if ($this->getAcknowledgement($licenseEntry) === self::ACK_CONCLUDED) {
152  $concludedLicenses = array_merge($concludedLicenses, $items);

154  $infoInFileLicenses = array_merge($infoInFileLicenses, $items);
155  }
156  }
157  }
158 
159  // CycloneDX 1.5+ evidence block
160  if (isset($component['evidence']) && is_array($component['evidence'])) {
161  $evidence = $component['evidence'];
162  if (isset($evidence['licenses']) && is_array($evidence['licenses'])) {
163  foreach ($evidence['licenses'] as $licenseEntry) {
164  $infoInFileLicenses = array_merge($infoInFileLicenses,
165  $this->parseLicenseEntry($licenseEntry));
166  }
167  }
168  if (isset($evidence['copyright']) && is_array($evidence['copyright'])) {
169  foreach ($evidence['copyright'] as $cp) {
170  if (isset($cp['text']) && is_string($cp['text'])) {
171  $copyrightTexts = array_merge($copyrightTexts,
172  $this->splitCopyrightText($cp['text']));
173  }
174  }
175  }
176  }
177 
178  // Fallback to top-level copyright if evidence is not present or empty
179  if (empty($copyrightTexts) && isset($component['copyright']) &&
180  is_string($component['copyright'])) {
181  $copyrightTexts = $this->splitCopyrightText($component['copyright']);
182  }
183 
184  return new ReportImportData($infoInFileLicenses, $concludedLicenses, $copyrightTexts);
185  }

245  private function parseLicenseObject($license)
246  {
247  $rawId = isset($license['id']) ? $license['id']
248  : (isset($license['name']) ? $license['name'] : null);
249  if ($rawId === null || $rawId === '') {
250  return [];
251  }
252 
253  $text = isset($license['text']['content']) ? $license['text']['content'] : '';
254  if ($text !== '' && isset($license['text']['encoding']) &&
255  strtolower($license['text']['encoding']) === 'base64') {
256  $decoded = base64_decode($text, true);
257  $text = ($decoded === false) ? '' : $decoded;
258  }
259  $url = isset($license['url']) ? $license['url'] : '';
260  $name = isset($license['name']) ? $this->stripLicenseRefPrefix($license['name']) : null;
261 
262  return [$this->createItem($rawId, $name, $text, $url)];
263  }

272  private function parseLicenseExpression($expression)
273  {
274  $tokens = array_filter(preg_split('/\s+/',
275  str_replace(['(', ')'], ' ', trim($expression))));
276  $items = [];
277  foreach ($tokens as $token) {
278  $upper = strtoupper($token);
279  if ($upper === 'OR') {
280  $items[] = new ReportImportDataItem("Dual-license");
281  } elseif ($upper === 'AND' || $upper === 'WITH') {
282  continue;
283  } else {
284  $items[] = $this->createItem($token);
285  }
286  }
287  return $items;
288  }

297  private function createItem($rawId, $name = null, $text = '', $url = '')
298  {
299  $isLicenseRef = strpos($rawId, \Fossology\Lib\Data\LicenseRef::SPDXREF_PREFIX) === 0;
300  $id = $this->stripLicenseRefPrefix($rawId);
301 
302  // FOSSology appends "-<md5>" to LicenseRefs carrying a custom text
303  $isCustomText = false;
304  if ($isLicenseRef && strlen($id) > 33 &&
305  substr($id, -33, 1) === "-" && ctype_alnum(substr($id, -32))) {
306  $id = substr($id, 0, -33);
307  $isCustomText = true;
308  }
309 
310  $item = new ReportImportDataItem($id);
311  if ($isCustomText && $text !== '') {
312  $item->setCustomText($text);
313  }
314  $item->setLicenseCandidate($name !== null ? $name : $id, $text, !$isLicenseRef, $url);
315  return $item;
316  }
