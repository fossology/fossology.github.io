90 static void make_tree(long count)
91 {
92  TreeSize = count;
93  Tree = calloc(count, sizeof(*Tree));
94  actual = calloc(count, sizeof(*actual));
95  expected = calloc(count, sizeof(*expected));
96  endpoints = calloc(2 * count + 1, sizeof(*endpoints));
97  /* A real successful PGresult lets fo_checkPQcommand() run unchanged. */

103  FO_ASSERT_PTR_NOT_NULL_FATAL(command_result);
104  for (long i = 0; i < count; ++i)
105  {
106  /* IDs are deliberately nonconsecutive and unrelated to traversal order. */
107  Tree[i].UploadtreePk = 1001 + 7 * (count - i);
108  Tree[i].Child = -1;
109  Tree[i].Sibling = -1;
110  }
111 }

113 static PGresult *capture_update(PGconn *connection, const char *query)
114 {
115  char table[128];
116  int end = 0;
117  FO_ASSERT_PTR_EQUAL(connection, pgConn);
118  FO_ASSERT_FATAL(update_count < TreeSize);
119  /* The previous update must be cleared and heartbeated before the next one. */
120  FO_ASSERT(clear_count == update_count);
121  FO_ASSERT(heart_count == update_count);
122  nested_set *row = &actual[update_count];
123  int fields = sscanf(query,
124  "UPDATE %127s SET lft='%ld', rgt='%ld' WHERE uploadtree_pk='%ld'%n",
125  table, &row->lft, &row->rgt, &row->pk, &end);
126  FO_ASSERT_EQUAL_FATAL(fields, 4);
127  FO_ASSERT(query[end] == '\0');
128  FO_ASSERT_STRING_EQUAL(table, uploadtree_tablename);
129  ++update_count;
130  return command_result;
131 }

142 static void capture_heart(int items)
143 {
144  FO_ASSERT_EQUAL(items, 1);
145  FO_ASSERT(clear_count == update_count);
146  FO_ASSERT(heart_count + 1 == update_count);
147  ++heart_count;
148 
149  /* Simulate the alive flag being reset between updates by the alarm handler.
150  * Each call must refresh it, not just increment the processed-item count.
151  */
152  alive = FALSE;
153  fo_scheduler_heart(items);
154  FO_ASSERT_TRUE(alive);
155  FO_ASSERT(g_atomic_int_get(&items_processed) == heart_count);
156 }

179 static void run_walk(void)
180 {
181  WalkTree(0, 0);
182  FO_ASSERT_FATAL(update_count == TreeSize);
183  FO_ASSERT(clear_count == TreeSize);
184  FO_ASSERT(heart_count == TreeSize);
185  FO_ASSERT(g_atomic_int_get(&items_processed) == TreeSize);
186  FO_ASSERT(SetNum == 2 * TreeSize);
187  qsort(actual, TreeSize, sizeof(*actual), compare_pk);
188 
189  for (long i = 0; i < TreeSize; ++i)
190  {
191  const nested_set *row = &actual[i];
192  FO_ASSERT_FATAL(row->lft >= 1 && row->rgt <= 2 * TreeSize);
193  FO_ASSERT_FATAL(row->lft < row->rgt);
194  FO_ASSERT_FALSE(endpoints[row->lft]);
195  FO_ASSERT_FALSE(endpoints[row->rgt]);
196  endpoints[row->lft] = endpoints[row->rgt] = 1;
197  }
198  for (long i = 1; i <= 2 * TreeSize; ++i)
199  {
200  FO_ASSERT_TRUE(endpoints[i]);
201  }
202  for (long i = 0; i < TreeSize; ++i)
203  {
204  const nested_set *parent = find_row(i);
205  FO_ASSERT_PTR_NOT_NULL_FATAL(parent);
206  for (long child = Tree[i].Child; child >= 0; child = Tree[child].Sibling)
207  {
208  const nested_set *row = find_row(child);
209  FO_ASSERT_PTR_NOT_NULL_FATAL(row);
210  FO_ASSERT(parent->lft < row->lft);
211  FO_ASSERT(row->rgt < parent->rgt);
212  }
213  }
214  assert_range(0, 1, 2 * TreeSize);
215 }

221 static void walk_recursive(long index, long *number)
222 {
223  long left = (*number)++;
224  if (Tree[index].Child > -1)
225  {
226  walk_recursive(Tree[index].Child, number);
227  ++*number;
228  }
229  expected[expected_count++] =
230  (nested_set){Tree[index].UploadtreePk, left, *number};
231  if (Tree[index].Sibling > -1)
232  {
233  ++*number;
234  walk_recursive(Tree[index].Sibling, number);
235  }
236 }
