13 class PythonParserTest(unittest.TestCase):
14  def test_null_project_urls_is_accepted(self):
15  components = [
16  {'name': name, 'version': '1.0', 'purl': f'pkg:pypi/{name}@1.0'}
17  for name in ['example-one', 'example-two']
18  ]
19  with tempfile.TemporaryDirectory() as directory:
20  sbom = os.path.join(directory, 'bom.json')
21  with open(sbom, 'w', encoding='utf-8') as stream:
22  json.dump({'components': components}, stream)
23  parser = Parser(sbom)
24  parser.classify_components(directory)
25  responses = []
26  for name in ['example-one', 'example-two']:
27  response = Mock()
28  response.json.return_value = {
29  'info': {'project_urls': None},
30  'urls': [{'packagetype': 'sdist',
31  'url': f'https://example.com/{name}.tar.gz'}],
32  }
33  responses.append(response)
34  with patch('ScanDeps.Parsers.requests.get', side_effect=responses):
35  with self.assertNoLogs(level='ERROR'):
36  PythonParser().parse_components(parser)
37  for component in parser.python_components:
38  name = component['name']
39  self.assertEqual(component[DOWNLOAD_URL_KEY],
40  f'https://example.com/{name}.tar.gz')
41  self.assertNotIn('vcs_url', component)
42  self.assertNotIn('homepage_url', component)

44  def test_project_urls_preserves_source_and_homepage(self):

47  with open(sbom, 'w', encoding='utf-8') as stream:
48  json.dump({'components': [{'name': 'example', 'version': '1.0',
49  'purl': 'pkg:pypi/example@1.0'}]}, stream)

52  response = Mock()
53  response.json.return_value = {
54  'info': {
55  'project_urls': {
56  'Source': 'https://example.com/source',
57  'Homepage': 'https://example.com',
58  },
59  },
60  'urls': [{'packagetype': 'sdist',
61  'url': 'https://example.com/example.tar.gz'}],
62  }
63  with patch('ScanDeps.Parsers.requests.get', return_value=response):
64  PythonParser().parse_components(parser)
65  component = parser.python_components[0]
66  self.assertEqual(component['vcs_url'], 'https://example.com/source')
