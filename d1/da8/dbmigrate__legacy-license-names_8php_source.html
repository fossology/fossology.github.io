151 function updateExcludedObligationNames(DbManager $dbManager, $old, $new)
152 {
153  if (!$dbManager->existsTable('report_info')) {
154  return;
155  }
156 
157  /* Match the escaping the column was written with, then the one the pattern
158  * needs. A literal backslash has to survive the replacement string too. */
159  $oldName = trim(json_encode($old), '"');
160  $newName = trim(json_encode($new), '"');
161  $pattern = '(^|[^A-Za-z0-9-])' . preg_quote($oldName) . '([^A-Za-z0-9-]|$)';
162  $replacement = '\1' . str_replace('\\', '\\\\', $newName) . '\2';
163 
164  legacyLicenseWrite($dbManager,
165  "UPDATE report_info
166  SET ri_excluded_obligations =
167  regexp_replace(ri_excluded_obligations::text, $1, $2, 'g')::json
168  WHERE ri_excluded_obligations::text ~ $1",
169  array($pattern, $replacement), __FUNCTION__);
170 }

180 function renameOrMergeLicense(DbManager $dbManager, $old, $new, $verbose = false)
181 {
182  $row = $dbManager->getSingleRow(
183  "SELECT
184  (SELECT rf_pk FROM license_ref WHERE rf_shortname=$1 LIMIT 1) AS old_id,
185  (SELECT rf_pk FROM license_ref WHERE rf_shortname=$2 LIMIT 1) AS new_id",
186  array($old, $new),
187  __FUNCTION__ . '.getIds'
188  );
189 
190  $oldId = (!empty($row) && !empty($row['old_id'])) ? intval($row['old_id']) : 0;
191  $newId = (!empty($row) && !empty($row['new_id'])) ? intval($row['new_id']) : 0;
192 
193  if ($oldId === 0 || $oldId === $newId) {
194  return;
195  }
196 
197  if ($newId === 0) {
198  legacyLicenseWrite($dbManager, "UPDATE license_ref SET rf_shortname=$1 WHERE rf_pk=$2",
199  array($new, $oldId), __FUNCTION__ . '.rename');
200  if ($verbose) {
201  print "renamed license $old to $new\n";
202  }
203  return;
204  }
205 
206  foreach (getLicenseRefReferences() as $ref) {
207  list ($table, $column) = $ref;
208  if (!$dbManager->existsTable($table)) {
209  continue;
210  }
211  legacyLicenseWrite($dbManager, "UPDATE $table SET $column=$1 WHERE $column=$2",
212  array($newId, $oldId), __FUNCTION__ . ".move.$table.$column");
213  }
214 
215  legacyLicenseWrite($dbManager, "DELETE FROM license_ref WHERE rf_pk=$1",
216  array($oldId), __FUNCTION__ . '.deleteLegacy');
217  if ($verbose) {
218  print "merged license $old into existing $new\n";
219  }
220 }

227 function Migrate_Legacy_License_Names(DbManager $dbManager, $verbose = false)
228 {
229  if (!$dbManager->existsTable('license_ref')) {

232
