82 function renameOrMergeLicense(DbManager $dbManager, $old, $new, $verbose = false)
83 {
84  $row = $dbManager->getSingleRow(
85  "SELECT
86  (SELECT rf_pk FROM license_ref WHERE rf_shortname=$1 LIMIT 1) AS old_id,
87  (SELECT rf_pk FROM license_ref WHERE rf_shortname=$2 LIMIT 1) AS new_id",
88  array($old, $new),
89  __FUNCTION__ . '.getIds'
90  );
91 
92  $oldId = (!empty($row) && !empty($row['old_id'])) ? intval($row['old_id']) : 0;
93  $newId = (!empty($row) && !empty($row['new_id'])) ? intval($row['new_id']) : 0;
94 
95  if ($oldId === 0 || $oldId === $newId) {
96  return;
97  }
98 
99  if ($newId === 0) {
100  legacyLicenseWrite($dbManager, "UPDATE license_ref SET rf_shortname=$1 WHERE rf_pk=$2",
101  array($new, $oldId), __FUNCTION__ . '.rename');
102  if ($verbose) {
103  print "renamed license $old to $new\n";
104  }
105  return;
106  }
107 
108  foreach (getLicenseRefReferences() as $ref) {
109  list ($table, $column) = $ref;
110  if (!$dbManager->existsTable($table)) {
111  continue;
112  }
113  legacyLicenseWrite($dbManager, "UPDATE $table SET $column=$1 WHERE $column=$2",
114  array($newId, $oldId), __FUNCTION__ . ".move.$table.$column");
115  }
116 
117  legacyLicenseWrite($dbManager, "DELETE FROM license_ref WHERE rf_pk=$1",
118  array($oldId), __FUNCTION__ . '.deleteLegacy');
119  if ($verbose) {
120  print "merged license $old into existing $new\n";
121  }
122 }
